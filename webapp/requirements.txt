# mosaic design webapp (host-side, runs on a login node).
#
# The app never imports mosaic and never runs a model — it reads files and
# queues SLURM jobs — so it needs only a plain Python 3.11+ environment:
#
#   module load python/3.13.12-fasrc01
#   uv venv webapp/.venv && uv pip install --python webapp/.venv -r webapp/requirements.txt
#
# See docs/WEBAPP.md.
# Charts use Altair, which Streamlit already bundles (Vega-Lite) — no plotly.
# plotly's wheel is ~45 MB and its full bundle is pushed to the browser on every
# chart; on a login node the app must stay light.
streamlit>=1.40

# st.dataframe and Altair serialise through Arrow; without these it raises
# ModuleNotFoundError: No module named 'pyarrow' at render time, not import time.
pyarrow>=15
pandas>=2
numpy>=2

# Docs tab (markdown -> HTML) and 3D structure views.
markdown>=3.5
py3Dmol>=2.0
gemmi>=0.6          # chain identification for the 3D complex view
